const { ReservasProvider, useReservas, STORAGE_KEY } = ReservasContextModule;
const setItemOriginal = Storage.prototype.setItem;

const wrapper = ({ children }) => <ReservasProvider>{children}</ReservasProvider>;

function mockearSetItemFalla() {
  return vi.spyOn(Storage.prototype, 'setItem').mockImplementation(() => {
    throw new DOMException('QuotaExceededError', 'QuotaExceededError');
  });
}

function leerGuardado() {
  const raw = window.localStorage.getItem(STORAGE_KEY);
  return raw === null ? null : JSON.parse(raw);
}

function escribirYDespacharStorage(datos) {
  const json = JSON.stringify(datos);
  setItemOriginal.call(window.localStorage, STORAGE_KEY, json);
  act(() => {
    window.dispatchEvent(
      new StorageEvent('storage', { key: STORAGE_KEY, newValue: json })
    );
  });
  return json;
}

function crearReservaExterna({
  id,
  codigo,
  canchaId = 'padel-1',
  fecha,
  horas,
  nombre = 'Otra Pestaña',
}) {
  const cancha = semilla.canchas.find((c) => c.id === canchaId);
  return {
    id,
    codigo,
    tipo: 'individual',
    canchaId,
    canchaNombre: cancha ? cancha.nombre : canchaId,
    deporte: cancha ? cancha.deporte : 'padel',
    fecha,
    hora: horas[0],
    horas: [...horas],
    nombre,
  };
}

function horaDisponible(result, canchaId, fecha, hora) {
  return result.current
    .disponibilidad(canchaId, fecha)
    .find((h) => h.hora === hora)?.disponible;
}

function assertSinAvisos(result) {
  expect(
    Array.isArray(result.current.avisosConflicto),
    'avisosConflicto debe existir y ser un arreglo'
  ).toBe(true);
  expect(result.current.avisosConflicto).toEqual([]);
}

function assertUnAvisoConCodigo(result, codigo) {
  expect(
    Array.isArray(result.current.avisosConflicto),
    'avisosConflicto debe existir y ser un arreglo'
  ).toBe(true);
  expect(result.current.avisosConflicto).toHaveLength(1);
  const aviso = result.current.avisosConflicto[0];
  expect(aviso.mensaje).toContain(codigo);
  expect(aviso.mensaje).not.toMatch(/localStorage|pestaña|storage/i);
  expect(
    aviso.codigo,
    'avisosConflicto[0].codigo debe coincidir con el código de la reserva'
  ).toBe(codigo);
  return aviso;
}

  it('N9: cancelar desde el flujo del chat (individual con plazo, evento rechazado y fuera de plazo)', () => {
    vi.useFakeTimers({ toFake: ['Date'] });
    vi.setSystemTime(new Date(2026, 10, 16, 10, 0, 0));

    const { result } = renderHook(() => useReservas(), { wrapper });
    const dia0 = result.current.dias[0]; // 2026-11-16 (14:00 está libre en padel-1 y a 4h < 6h)
    const dia3 = result.current.dias[3]; // 2026-11-19
    const dia4 = result.current.dias[4]; // 2026-11-20

    let resInd;
    let resEv;
    let resCerca;
    act(() => {
      resInd = result.current.reservar({
        canchaId: 'padel-1',
        fecha: dia3,
        horas: ['16:00'],
        nombre: 'Ana Con Plazo',
        telefono: '3001112233',
      });
      resEv = result.current.reservarEvento({
        subtipo: 'cumpleanos',
        fecha: dia4,
        canchaIds: ['voley-1'],
        horas: ['09:00', '10:00'],
        invitados: 12,
        nombre: 'Cumple N9',
      });
      resCerca = result.current.reservar({
        canchaId: 'padel-1',
        fecha: dia0,
        horas: ['14:00'],
        nombre: 'Carlos Sin Plazo',
        telefono: '3004445566',
      });
    });
    expect(resInd.ok).toBe(true);
    expect(resEv.ok).toBe(true);
    expect(resCerca.ok).toBe(true);

    const idInd = result.current.reservas.find((r) => r.codigo === resInd.codigo).id;
    const idEv = result.current.reservas.find((r) => r.codigo === resEv.codigo).id;
    const idCerca = result.current.reservas.find((r) => r.codigo === resCerca.codigo).id;

    // 1. Cancelar individual con plazo suficiente -> ok y hora libre
    let cancInd;
    act(() => {
      cancInd = result.current.cancelar(idInd);
    });
    expect(cancInd).toEqual({ ok: true });
    expect(result.current.reservas.some((r) => r.id === idInd)).toBe(false);
    expect(horaDisponible(result, 'padel-1', dia3, '16:00')).toBe(true);

    const guardadoTrasInd = leerGuardado();

    // 2. Cancelar evento -> error de evento y no toca nada
    let cancEv;
    act(() => {
      cancEv = result.current.cancelar(idEv);
    });
    expect(cancEv).toEqual({
      ok: false,
      error: cancelacion.MENSAJES_CANCELACION.evento,
    });
    expect(result.current.reservas.some((r) => r.id === idEv)).toBe(true);

    expect(horaDisponible(result, 'voley-1', dia4, '10:00')).toBe(false);
    expect(leerGuardado()).toEqual(guardadoTrasInd);

    let cancCerca;
    act(() => {
      cancCerca = result.current.cancelar(idCerca);
    });
    expect(cancCerca).toEqual({
      ok: false,
      error: cancelacion.MENSAJES_CANCELACION.fueraDePlazo,
    });
    expect(result.current.reservas.some((r) => r.id === idCerca)).toBe(true);
    expect(horaDisponible(result, 'padel-1', dia0, '14:00')).toBe(false);
    expect(leerGuardado()).toEqual(guardadoTrasInd);
  });
