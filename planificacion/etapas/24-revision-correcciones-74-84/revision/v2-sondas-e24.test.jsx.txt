const setItemOriginal = Storage.prototype.setItem;

const wrapper = ({ children }) => <ReservasProvider>{children}</ReservasProvider>;

function mockearSetItemFalla() {
  return vi.spyOn(Storage.prototype, 'setItem').mockImplementation(() => {
    throw new DOMException('QuotaExceededError', 'QuotaExceededError');
  });
}

function leerGuardado() {
  const raw = window.localStorage.getItem(STORAGE_KEY);
  return raw === null ? null : JSON.parse(raw);
}

function escribirYDespacharStorage(datos) {
  const json = JSON.stringify(datos);
  setItemOriginal.call(window.localStorage, STORAGE_KEY, json);
  act(() => {
    window.dispatchEvent(
      new StorageEvent('storage', { key: STORAGE_KEY, newValue: json })
    );
  });
  return json;
}

function crearReservaExterna({
  id,
  codigo,
  canchaId = 'padel-1',
  fecha,
  horas,
  nombre = 'Otra Pestaña',
  telefono = null,
}) {
  const cancha = semilla.canchas.find((c) => c.id === canchaId);
  return {
    id,
    codigo,
    tipo: 'individual',
    canchaId,
    canchaNombre: cancha ? cancha.nombre : canchaId,
    deporte: cancha ? cancha.deporte : 'padel',
    fecha,
    hora: horas[0],
    horas: [...horas],
    nombre,
    telefono,
  };
}

function horaDisponible(result, canchaId, fecha, hora) {
  return result.current
    .disponibilidad(canchaId, fecha)
    .find((h) => h.hora === hora)?.disponible;
}

function CapturarContextos({ onApiListo, onVistaListo }) {
  const api = useReservas();
  const vista = useVistaCanchas();
  const listoRef = useRef(false);
  useEffect(() => {
    if (!listoRef.current) {
      listoRef.current = true;
      onApiListo?.(api);
      onVistaListo?.(vista);
    }
  }, [api, vista, onApiListo, onVistaListo]);
  return null;
}

function renderVistaConAviso(ruta = '/reservar/padel', { onApiListo, onVistaListo } = {}) {
  window.history.pushState({}, '', ruta);
  return render(
    <ReservasProvider>
      <VistaCanchasProvider>
        <AvisoConflictoReserva />
        <VistaCanchas />
        {(onApiListo || onVistaListo) && (
          <CapturarContextos onApiListo={onApiListo} onVistaListo={onVistaListo} />
        )}
      </VistaCanchasProvider>
    </ReservasProvider>
  );
}

function despacharChoqueStorage({
  base,
  canchaId,
  fecha,
  hora,
  codigoExterno = 'SPF-EXT99',
  reservasExtra = [],
}) {
  const otroEstado = structuredClone(base);
  reservasExtra.forEach((r) => {
    const f = r.fecha;
    const c = r.canchaId;
    r.horas.forEach((h) => {
      if (!otroEstado.ocupados[c][f].includes(h)) {
        otroEstado.ocupados[c][f].push(h);
      }
    });
    otroEstado.reservas.push(r);
  });
  if (!otroEstado.ocupados[canchaId][fecha].includes(hora)) {
    otroEstado.ocupados[canchaId][fecha].push(hora);
  }
  const rb = crearReservaExterna({
    id: `${canchaId}__${fecha}__${hora}__9999_${codigoExterno}`,
    codigo: codigoExterno,
    canchaId,
    fecha,
    horas: [hora],
  });
  otroEstado.reservas.push(rb);
  const jsonOtro = JSON.stringify(otroEstado);
  setItemOriginal.call(window.localStorage, STORAGE_KEY, jsonOtro);
  act(() => {
    window.dispatchEvent(
      new StorageEvent('storage', { key: STORAGE_KEY, newValue: jsonOtro })
    );
  });
  return rb;
}
